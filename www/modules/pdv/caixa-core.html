/** * Vancely Software - Caixa Core (caixa-core.js) * Script principal de controle do Frente de Caixa (PDV) */ // Estado global da aplicação do Caixa const estadoCaixa = { aberto: false, operador: "Ativo", caixaId: "#01", carrinho: [], subtotal: 0, descontos: 0, totalGeral: 0 }; document.addEventListener('DOMContentLoaded', () => { console.log('Vancely Software - Frente de Caixa inicializado.'); atualizarInterfaceCaixa(); // Focar automaticamente no input de busca ao iniciar const inputBusca = document.getElementById('inputBusca'); if (inputBusca) { inputBusca.focus(); } }); // ========================================== // FUNÇÕES DE GESTÃO DO CAIXA (HEADER) // ========================================== window.acionarAbrirCaixa = function() { estadoCaixa.aberto = true; atualizarInterfaceCaixa(); alert('Caixa aberto com sucesso!'); }; window.acionarFecharCaixa = function() { if (confirm('Deseja realmente fechar o caixa atual?')) { estadoCaixa.aberto = false; atualizarInterfaceCaixa(); alert('Caixa fechado.'); } }; function atualizarInterfaceCaixa() { const txtStatus = document.getElementById('txtStatusCaixaHeader'); const btnAbrir = document.getElementById('btnAbrirCaixaHeader'); const btnFechar = document.getElementById('btnFecharCaixaHeader'); if (estadoCaixa.aberto) { if (txtStatus) { txtStatus.className = 'text-xs text-emerald-400 font-semibold flex items-center gap-1.5'; txtStatus.innerHTML = ' Caixa Aberto'; } if (btnAbrir) btnAbrir.classList.add('opacity-50', 'pointer-events-none'); if (btnFechar) btnFechar.classList.remove('opacity-50', 'pointer-events-none'); } else { if (txtStatus) { txtStatus.className = 'text-xs text-amber-400 font-semibold flex items-center gap-1.5'; txtStatus.innerHTML = ' Caixa Fechado (Necessário Abertura)'; } if (btnAbrir) btnAbrir.classList.remove('opacity-50', 'pointer-events-none'); if (btnFechar) btnFechar.classList.add('opacity-50', 'pointer-events-none'); } } // ========================================== // FUNÇÕES DE BUSCA E CARRINHO // ========================================== window.aoDigitarBusca = function(event) { const termo = event.target.value; const containerSugestoes = document.getElementById('sugestoesBusca'); if (!termo || termo.trim() === '') { if (containerSugestoes) containerSugestoes.classList.add('hidden'); return; } // Simulação de sugestões ou busca dinâmica if (containerSugestoes) { containerSugestoes.innerHTML = `
Produto Exemplo (${termo}) R$ 10,00
`; containerSugestoes.classList.remove('hidden'); } }; window.tratarEnterBuscaCaixa = function(event) { if (event.key === 'Enter') { event.preventDefault(); window.adicionarItemExemplo(); } }; window.adicionarItemExemplo = function() { const item = { id: Date.now(), nome: 'Produto Exemplo de Catálogo', preco: 10.00, quantidade: 1 }; estadoCaixa.carrinho.push(item); recalcularCarrinho(); const inputBusca = document.getElementById('inputBusca'); if (inputBusca) { inputBusca.value = ''; inputBusca.focus(); } const containerSugestoes = document.getElementById('sugestoesBusca'); if (containerSugestoes) containerSugestoes.classList.add('hidden'); }; function recalcularCarrinho() { const listaVazia = document.getElementById('listaCarrinhoVazio'); const listaItens = document.getElementById('listaCarrinhoItens'); const txtSubtotal = document.getElementById('txtResumoSubtotal'); const txtTotalGeral = document.getElementById('txtResumoTotalGeral'); if (estadoCaixa.carrinho.length === 0) { if (listaVazia) listaVazia.classList.remove('hidden'); if (listaItens) { listaItens.classList.add('hidden'); listaItens.innerHTML = ''; } estadoCaixa.subtotal = 0; estadoCaixa.totalGeral = 0; } else { if (listaVazia) listaVazia.classList.add('hidden'); if (listaItens) { listaItens.classList.remove('hidden'); listaItens.innerHTML = estadoCaixa.carrinho.map((prod, index) => `
${prod.nome}
${prod.quantidade}x R$ ${prod.preco.toFixed(2)}
R$ ${(prod.quantidade * prod.preco).toFixed(2)}
`).join(''); } estadoCaixa.subtotal = estadoCaixa.carrinho.reduce((acc, p) => acc + (p.preco * p.quantidade), 0); estadoCaixa.totalGeral = estadoCaixa.subtotal - estadoCaixa.descontos; } if (txtSubtotal) txtSubtotal.innerText = `R$ ${estadoCaixa.subtotal.toFixed(2)}`; if (txtTotalGeral) txtTotalGeral.innerText = `R$ ${estadoCaixa.totalGeral.toFixed(2)}`; } window.abrirCameraScanner = function() { alert('Módulo de leitor contínuo por câmera acionado.'); }; // ========================================== // FUNÇÕES DE ATALHOS (F1 a F12) // ========================================== window.acaoAtalhoF1 = function() { console.log('Atalhos: F1 acionado (CPF na Nota)'); const cpf = prompt('Digite o CPF/CNPJ para a Nota Fiscal:'); if (cpf) alert(`CPF ${cpf} vinculado à venda com sucesso!`); }; window.acaoAtalhoF2 = function() { console.log('Atalhos: F2 acionado (Vendedor)'); const vendedor = prompt('Informe o código ou nome do vendedor:'); if (vendedor) alert(`Vendedor ${vendedor} vinculado.`); }; window.acaoAtalhoF3 = function() { console.log('Atalhos: F3 acionado (Dinheiro/Pix Rápido)'); if (estadoCaixa.carrinho.length === 0) { alert('Adicione itens ao carrinho antes de finalizar.'); return; } alert(`Pagamento em Dinheiro/Pix Rápido processado para o total de R$ ${estadoCaixa.totalGeral.toFixed(2)}`); estadoCaixa.carrinho = []; recalcularCarrinho(); }; window.acaoAtalhoF4 = function() { console.log('Atalhos: F4 acionado (Cartão Débito)'); if (estadoCaixa.carrinho.length === 0) { alert('Adicione itens ao carrinho antes de finalizar.'); return; } alert('Transação de Débito acionada no terminal.'); estadoCaixa.carrinho = []; recalcularCarrinho(); }; window.acaoAtalhoF5 = function() { console.log('Atalhos: F5 acionado (Pesquisa / Focar Busca)'); const inputBusca = document.getElementById('inputBusca'); if (inputBusca) { inputBusca.focus(); inputBusca.select(); } }; window.acaoAtalhoF6 = function() { console.log('Atalhos: F6 acionado (Pausar Venda)'); alert('Venda atual pausada/colocada em espera.'); estadoCaixa.carrinho = []; recalcularCarrinho(); }; window.acionarFinalizarVenda = function() { console.log('Atalhos: F7 acionado (Finalizar Venda / Checkout)'); if (estadoCaixa.carrinho.length === 0) { alert('Carrinho vazio. Impossível finalizar.'); return; } alert(`Checkout aberto para o valor total de R$ ${estadoCaixa.totalGeral.toFixed(2)}`); estadoCaixa.carrinho = []; recalcularCarrinho(); }; window.acaoAtalhoF8 = function() { console.log('Atalhos: F8 acionado (Parcelado / Crédito)'); if (estadoCaixa.carrinho.length === 0) { alert('Carrinho vazio.'); return; } alert('Abertura de modal de parcelamento no cartão de crédito.'); }; window.acaoAtalhoF9 = function() { console.log('Atalhos: F9 acionado (Movimento de Caixa - Sangria/Suprimento)'); const tipo = prompt('Digite "1" para Suprimento ou "2" para Sangria:'); if (tipo === '1') { prompt('Valor do Suprimento (R$):'); alert('Suprimento registrado com sucesso.'); } else if (tipo === '2') { prompt('Valor da Sangria (R$):'); alert('Sangria registrada com sucesso.'); } }; window.acaoAtalhoF10 = function() { console.log('Atalhos: F10 acionado (Consulta de Preço)'); const codigo = prompt('Informe o código de barras para consultar o preço:'); if (codigo) { alert(`O produto do código ${codigo} custa R$ 15,00 (Exemplo).`); } }; window.acaoAtalhoF11 = function() { console.log('Atalhos: F11 acionado (Cancelar Item)'); if (estadoCaixa.carrinho.length === 0) { alert('Nenhum item no carrinho para cancelar.'); return; } const pin = prompt('PIN do Administrador necessário para cancelar o item:'); if (pin === '1234') { estadoCaixa.carrinho.pop(); recalcularCarrinho(); alert('Último item cancelado com sucesso.'); } else if (pin !== null) { alert('PIN incorreto!'); } }; window.acaoAtalhoF12 = function() { console.log('Atalhos: F12 acionado (Cancelar Venda)'); if (estadoCaixa.carrinho.length === 0) { alert('Nenhum item para cancelar.'); return; } const pin = prompt('PIN do Administrador necessário para cancelar a venda inteira:'); if (pin === '1234') { estadoCaixa.carrinho = []; recalcularCarrinho(); alert('Venda cancelada totalmente.'); } else if (pin !== null) { alert('PIN incorreto!'); } }; window.acaoAtalhoParcelamento = function() { window.acaoAtalhoF8(); }; // ========================================== // LISTENER GLOBAL DE ATALHOS DE TECLADO (F1 a F12) // ========================================== window.addEventListener('keydown', (e) => { // Verifica se a tecla pressionada é do tipo F1 até F12 if (e.key >= 'F1' && e.key <= 'F12') { e.preventDefault(); // Impede o comportamento padrão do browser (ex: ajuda do F1) switch (e.key) { case 'F1': if (typeof window.acaoAtalhoF1 === 'function') window.acaoAtalhoF1(); break; case 'F2': if (typeof window.acaoAtalhoF2 === 'function') window.acaoAtalhoF2(); break; case 'F3': if (typeof window.acaoAtalhoF3 === 'function') window.acaoAtalhoF3(); break; case 'F4': if (typeof window.acaoAtalhoF4 === 'function') window.acaoAtalhoF4(); break; case 'F5': if (typeof window.acaoAtalhoF5 === 'function') window.acaoAtalhoF5(); break; case 'F6': if (typeof window.acaoAtalhoF6 === 'function') window.acaoAtalhoF6(); break; case 'F7': if (typeof window.acionarFinalizarVenda === 'function') window.acionarFinalizarVenda(); break; case 'F8': if (typeof window.acaoAtalhoF8 === 'function') window.acaoAtalhoF8(); break; case 'F9': if (typeof window.acaoAtalhoF9 === 'function') window.acaoAtalhoF9(); break; case 'F10': if (typeof window.acaoAtalhoF10 === 'function') window.acaoAtalhoF10(); break; case 'F11': if (typeof window.acaoAtalhoF11 === 'function') window.acaoAtalhoF11(); break; case 'F12': if (typeof window.acaoAtalhoF12 === 'function') window.acaoAtalhoF12(); break; } } });
